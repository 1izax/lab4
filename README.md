# Домашнее задание к работе 4
## Условие задачи
На развилке дорог стоит знак. Он показывает направление "Налево", если только одно из двух чисел (A или B), полученных от датчиков traffic flow, является четным. Запишите условие для показа направления "Налево".
## 1. Алгоритм и блок-схема
### Алгоритм решения
1. Начало
2. Объявить константу и переменные:
   - число от первого датчика: A
   - число от второго датчика: B
3. Ввести значение переменной:
   - ввод значения A
   - ввод значения B
4. Проверить условие (ровно одно число чётное):
   - если A % 2 != B % 2, то направление = "Налево"
   - иначе направление = "Направо"
5. Вывести результаты расчетов с подстановкой всех значений в текст
6. Конец

## 2. Реализация программы
```
#include <stdio.h>
#include <locale.h>

int road()
{
	setlocale(LC_CTYPE, "RUS");

	int A, B;

	printf("=============================================================\n");
	printf("Условие задачи:\n");
	printf("На развилке дорог установлен знак. Он указывает\n");
	printf("направление \"Налево\" только в том случае, когда ровно\n");
	printf("одно из двух чисел (A или B), полученных от датчиков\n");
	printf("traffic flow, окажется чётным. Во всех остальных\n");
	printf("случаях знак указывает направление \"Направо\".\n");
	printf("=============================================================\n");
	printf("Введите показания датчика A:\n");
	scanf("%d", &A);
	printf("Введите показания датчика B:\n");
	scanf("%d", &B);
	printf("=============================================================\n");

	int pa = (A % 2 == 0);
	int pb = (B % 2 == 0);
	int goLeft = (pa != pb);

	printf("Исходные данные:\n");
	printf("Датчик A = %d (%s)\n", A, pa ? "чётное" : "нечётное");
	printf("Датчик B = %d (%s)\n", B, pb ? "чётное" : "нечётное");
	printf("=============================================================\n");
	printf("Результат:\n");
	if (goLeft)
	{
		printf("Ровно одно из чисел чётное.\n");
		printf("Направление: НАЛЕВО\n");
	}
	else
	{
		printf("Числа имеют одинаковую чётность.\n");
		printf("Направление: НАПРАВО\n");
	}
	printf("=============================================================\n");
}

int main()
{
	road();
	getchar();
	return 0;
}
```
